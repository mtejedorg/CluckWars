using UnityEngine; using UnityEngine.UIElements; using System.Linq; using System.Reflection; using System.Collections.Generic; using System.Text; using System.IO;
using CluckWars.Audio; using CluckWars.UI; using CluckWars.Gameplay;
// Phase 3B review probe: the in-game intro countdown (MatchOverlaysController.RefreshIntro). className "Intro".
public class Intro {
  sealed class RecAudio : IAudioService {
    public void PlaySFX(AudioClip clip, float volume = 1f, float pitch = 1f, float delaySeconds = 0f) { Lines.Add("      AUDIO t=" + (Time.unscaledTime - Start).ToString("0.000") + " " + (clip == null ? "null" : clip.name) + " pitch=" + pitch.ToString("0.00")); }
    public void PlayMusic(AudioClip clip, float volume = 1f, bool loop = true, float fadeInSeconds = 0f) { }
    public void FadeOutMusic(float seconds) { }
    public void SetMusicLevel(float level01, float seconds) { }
    public void CrossfadeMusic(AudioClip clip, float seconds) { }
    public void StopMusic() { }
    public bool IsMusicPlaying(AudioClip clip) { return false; }
    public void SetMasterVolume(float volume01) { }
  }
  static List<string> Lines { get { var l = System.AppDomain.CurrentDomain.GetData("intro.lines") as List<string>; if (l == null) { l = new List<string>(); System.AppDomain.CurrentDomain.SetData("intro.lines", l); } return l; } }
  static float Start { get { var o = System.AppDomain.CurrentDomain.GetData("intro.start"); return o == null ? 0f : (float)o; } set { System.AppDomain.CurrentDomain.SetData("intro.start", value); } }
  static string Prefix; static float Duration; static string LastText = ""; static bool Shot; static float ShotAt = -1; static IVisualElementScheduledItem Item;
  static string F(float v) { return v.ToString("0.00", System.Globalization.CultureInfo.InvariantCulture); }
  static UIDocument Doc() { return Object.FindObjectsByType<UIDocument>(FindObjectsSortMode.None).FirstOrDefault(x => x.rootVisualElement != null && x.rootVisualElement.Q<Label>("IntroNumber") != null); }

  static string ArmPrefix, ArmDur;
  static void Poll() { if (Doc() == null) return; var r = Install(ArmPrefix, ArmDur); if (r == "installed") UnityEditor.EditorApplication.update -= Poll; }
  // Install the moment the Game scene's overlay document exists (polling from the Bootstrap menu, before START is pressed).
  // MenuAudio is a project-scoped singleton: swap ITS service + catalogue so every cue from now on is recorded, whenever the overlays bind.
  public static string Arm(string prefix, string dur) {
    var ma = Zenject.ProjectContext.Instance.Container.Resolve<MenuAudio>();
    var cat = ScriptableObject.CreateInstance<UiAudioSO>();
    foreach (var f in typeof(UiAudioSO).GetFields()) if (f.FieldType == typeof(UiCue)) { var cue = (UiCue)f.GetValue(cat); cue.Clip = AudioClip.Create(f.Name, 64, 1, 8000, false); f.SetValue(cat, cue); }
    typeof(MenuAudio).GetField("_audio", BindingFlags.NonPublic | BindingFlags.Instance).SetValue(ma, new RecAudio());
    typeof(MenuAudio).GetField("_catalog", BindingFlags.NonPublic | BindingFlags.Instance).SetValue(ma, cat);
    Start = Time.unscaledTime; Lines.Add("=== armed (cues recorded from here; t relative to arm)");
    ArmPrefix = prefix; ArmDur = dur; UnityEditor.EditorApplication.update += Poll; return "armed"; }
  public static string Ready() { var d = Doc(); return d == null ? "no" : "yes"; }
  public static string Install(string prefix, string dur) {
    var d = Doc(); if (d == null) return "NODOC";
    var ctl = d.GetComponent<MatchOverlaysController>();
    var cat = ScriptableObject.CreateInstance<UiAudioSO>();
    foreach (var f in typeof(UiAudioSO).GetFields()) if (f.FieldType == typeof(UiCue)) { var cue = (UiCue)f.GetValue(cat); cue.Clip = AudioClip.Create(f.Name, 64, 1, 8000, false); f.SetValue(cat, cue); }
    Prefix = prefix; Duration = float.Parse(dur, System.Globalization.CultureInfo.InvariantCulture) + (Time.unscaledTime - Start); Shot = false; ShotAt = -1; LastText = "";
    Lines.Add("=== intro probe, allowed=" + MenuJuice.Allowed);
    var root = d.rootVisualElement;
    Item = root.schedule.Execute(() => {
      float t = Time.unscaledTime - Start; var n = root.Q<Label>("IntroNumber"); var ov = root.Q<VisualElement>("IntroOverlay");
      bool shown = ov.resolvedStyle.display == DisplayStyle.Flex;
      var gm = GameManager.Instance;
      if (shown) {
        Lines.Add(t.ToString("0.000", System.Globalization.CultureInfo.InvariantCulture) + " | IntroNumber '" + n.text + "' sc=" + F(n.resolvedStyle.scale.value.x) + " | remaining=" + (gm == null ? "?" : F(gm.IntroRemaining)));
        if (n.text != LastText) { LastText = n.text; if (n.text == "2" && !Shot) ShotAt = t + 0.10f; }
        if (ShotAt > 0 && !Shot && t >= ShotAt) { Shot = true; ScreenCapture.CaptureScreenshot(Prefix + "_mid_" + n.text + ".png"); Lines.Add("      SHOT (text=" + n.text + " sc=" + F(n.resolvedStyle.scale.value.x) + ")"); }
      }
      if (t > Duration) { Item.Pause(); Lines.Add("=== end"); }
    }).Every(1);
    return "installed";
  }
  public static string SetReduced(string on) { CluckWars.Settings.PlayerPreferences.ReducedMotionEnabled = on == "1"; return "reduced=" + CluckWars.Settings.PlayerPreferences.ReducedMotionEnabled; }
  public static string ClearReduced() { PlayerPrefs.DeleteKey(CluckWars.Settings.PlayerPreferences.ReducedMotionKey); PlayerPrefs.Save(); CluckWars.Settings.PlayerPreferences.ResetCache(); return "cleared"; }
  public static string Win() { var b = PlayerBase.ActiveBases.FirstOrDefault(x => x != null && x.IsClaimed); if (b == null) return "no claimed base"; b.FoodTotal = 999f; return "ok"; }
  public static string State() { var gm = GameManager.Instance; return gm == null ? "nogm" : gm.State + (gm.IsIntroActive ? "+intro" : ""); }
  public static string Dump(string path) { File.WriteAllText(path, string.Join("\n", Lines)); var n = Lines.Count; Lines.Clear(); return "wrote " + n; }
  public static string Press(string name) { var r = Object.FindObjectsByType<UIDocument>(FindObjectsSortMode.None).Select(x => x.rootVisualElement).FirstOrDefault(x => x != null && x.Q<Button>(name) != null); if (r == null) return "NOBTN " + name;
    var b = r.Q<Button>(name); var c = typeof(Button).GetField("m_Clickable", BindingFlags.NonPublic | BindingFlags.Instance).GetValue(b);
    ((System.Action)typeof(Clickable).GetField("clicked", BindingFlags.NonPublic | BindingFlags.Instance).GetValue(c)).Invoke(); return "clicked " + name; }
  public static string FillQuiet() { var r = Object.FindObjectsByType<UIDocument>(FindObjectsSortMode.None).Select(x => x.rootVisualElement).First(x => x != null && x.Q<Button>("ReadyBtn") != null); int taps = 0;
    for (int i = 0; i < 14; i++) { var rb = r.Q<Button>("ReadyBtn"); if (rb != null && rb.enabledSelf) break;
      var card = r.Query<VisualElement>(className: "cw-ability-card").ToList().FirstOrDefault(c => !c.ClassListContains("cw-ability-card--picked")); if (card == null) break;
      using (var e = ClickEvent.GetPooled()) { e.target = card; card.SendEvent(e); } taps++; }
    return "taps=" + taps; }

  // ---- Re-audit item 4 probe (python intro_probe.py <out> --item4) -----------------------------------
  // One EditorApplication.update hook per session: the menu's GET READY card (captured once fully faded in),
  // the scene swap, every GameManager state change, every intro-text change (3 / 2 / 1 / GO! captured once
  // per round), the GameManager's own audio (forwarded to the real service, so the session still sounds),
  // and the comeback banner (+ the music sources after GO and after the event). Scores are capped below the goal so the round ends on the TIMER.
  sealed class FwdAudio : IAudioService {
    public IAudioService Inner; public System.Action<string> Log;
    public void PlaySFX(AudioClip clip, float volume = 1f, float pitch = 1f, float delaySeconds = 0f) { Log("AUDIO sfx " + (clip == null ? "null" : clip.name)); Inner.PlaySFX(clip, volume, pitch, delaySeconds); }
    public void PlayMusic(AudioClip clip, float volume = 1f, bool loop = true, float fadeInSeconds = 0f) { Log("AUDIO music " + (clip == null ? "null" : clip.name)); Inner.PlayMusic(clip, volume, loop, fadeInSeconds); }
    public void FadeOutMusic(float seconds) { Inner.FadeOutMusic(seconds); }
    public void SetMusicLevel(float level01, float seconds) { Log("AUDIO musicLevel " + level01); Inner.SetMusicLevel(level01, seconds); }
    public void CrossfadeMusic(AudioClip clip, float seconds) { Log("AUDIO crossfade -> " + (clip == null ? "null" : clip.name)); Inner.CrossfadeMusic(clip, seconds); }
    public void StopMusic() { Log("AUDIO stopMusic"); Inner.StopMusic(); }
    public bool IsMusicPlaying(AudioClip clip) { return Inner.IsMusicPlaying(clip); }
    public void SetMasterVolume(float volume01) { Inner.SetMasterVolume(volume01); }
  }
  sealed class P4 {
    public string Out; public float T0, End; public int Round; public bool CardSeen, CardShot; public string Scene = "", Text = null, StateStr = "";
    public float ShotAt = -1; public string ShotName; public HashSet<string> Shot = new HashSet<string>(); public int Clamps; public bool BannerSeen;
    public bool GetReadyOn; public float GetReadyAt; public float BannerAt = -1; public bool MusicLogged;
    public static string Music() {
      var host = GameObject.Find("AudioServiceHost"); if (host == null) return "no AudioServiceHost";
      return string.Join(", ", host.GetComponents<AudioSource>().Where(a => a.isPlaying && a.loop).Select(a => (a.clip == null ? "null" : a.clip.name) + " vol=" + a.volume.ToString("0.00") + " t=" + a.time.ToString("0.00"))); }
    static string N(float v) { return v.ToString("0.000", System.Globalization.CultureInfo.InvariantCulture); }
    public void L(string m) { Lines.Add(N(Time.realtimeSinceStartup - T0) + " f" + Time.frameCount + " dt=" + N(Time.unscaledDeltaTime) + " | " + m); }
    public void Tick() {
      float t = Time.realtimeSinceStartup - T0;
      if (t > End) { UnityEditor.EditorApplication.update -= Tick; L("=== probe end"); return; }
      var scene = UnityEngine.SceneManagement.SceneManager.GetActiveScene().name;
      if (scene != Scene) { Scene = scene; L("SCENE " + scene); }
      var docs = Object.FindObjectsByType<UIDocument>(FindObjectsSortMode.None);
      foreach (var d in docs) {
        var root = d.rootVisualElement; if (root == null) continue;
        var card = root.Q<VisualElement>("GetReadyCard");
        if (card != null && card.resolvedStyle.display == DisplayStyle.Flex) {
          if (!CardSeen) { CardSeen = true; L("GETREADY card shown, opacity=" + N(card.resolvedStyle.opacity)); }
          if (!CardShot && card.resolvedStyle.opacity >= 0.99f) { CardShot = true; ScreenCapture.CaptureScreenshot(Out + "/getready_card.png"); L("SHOT getready_card.png (opacity " + N(card.resolvedStyle.opacity) + ")"); }
        }
      }
      var gm = GameManager.Instance;
      if (gm != null) {
        var f = typeof(GameManager).GetField("_audio", BindingFlags.NonPublic | BindingFlags.Instance);
        var a = f.GetValue(gm) as IAudioService;
        if (a != null && !(a is FwdAudio)) { f.SetValue(gm, new FwdAudio { Inner = a, Log = L }); L("GameManager audio wrapped"); }
        string st = gm.State + (gm.IsIntroActive ? "+intro" : "");
        if (st != StateStr) { if (gm.State == MatchState.Active && !StateStr.StartsWith("Active")) Round++; StateStr = st; L("STATE " + st + " round=" + Round + " timeRemaining=" + N(gm.TimeRemaining) + " introRemaining=" + N(gm.IntroRemaining)); }
        if (gm.State == MatchState.Active) {
          int cap = gm.FoodTargetToWin - 5;
          foreach (var b in PlayerBase.ActiveBases) if (b != null && b.FoodTotal > cap) { b.FoodTotal = cap; if (Clamps++ == 0) L("score cap engaged (" + cap + ") so the round ends on the timer"); }
        }
      }
      foreach (var d in docs) {
        var root = d.rootVisualElement; if (root == null) continue;
        var n = root.Q<Label>("IntroNumber"); var ov = root.Q<VisualElement>("IntroOverlay"); if (n == null || ov == null) continue;
        bool shown = ov.resolvedStyle.display == DisplayStyle.Flex;
        var gr = root.Q<VisualElement>("IntroGetReady");
        bool grOn = shown && gr != null && gr.resolvedStyle.display == DisplayStyle.Flex;
        if (grOn != GetReadyOn) {
          GetReadyOn = grOn;
          if (grOn) { GetReadyAt = Time.realtimeSinceStartup; L("INGAME GETREADY shown"); }
          else L("INGAME GETREADY hidden after " + N(Time.realtimeSinceStartup - GetReadyAt) + "s (intro text '" + n.text + "')");
        }
        string txt = shown ? n.text : "<hidden>";
        if (txt != Text) {
          Text = txt; L("INTRO '" + txt + "' state=" + (gm == null ? "nogm" : gm.State.ToString()) + " introRemaining=" + (gm == null ? "?" : N(gm.IntroRemaining)));
          string key = txt == "GO!" ? "go" : txt;
          if (txt == "<hidden>" && gm != null && gm.State == MatchState.Active && !gm.IsIntroActive) L("MUSIC after GO: " + Music());
          if ((txt == "3" || txt == "2" || txt == "1" || txt == "GO!") && Shot.Add(Round + key)) { ShotAt = Time.realtimeSinceStartup + (txt == "GO!" ? 0.18f : 0.30f); ShotName = (Round <= 1 ? "intro_" : "intro_r" + Round + "_") + key + ".png"; }
          else if (txt == "" && Shot.Add(Round + "ready")) { ShotAt = Time.realtimeSinceStartup + 0.15f; ShotName = "ingame_getready" + (Round == 0 ? "" : "_r" + (Round + 1)) + ".png"; }
        }
        if (ShotAt > 0 && Time.realtimeSinceStartup >= ShotAt) { ShotAt = -1; ScreenCapture.CaptureScreenshot(Out + "/" + ShotName); L("SHOT " + ShotName + " (text '" + n.text + "' shown=" + shown + ")"); }
      }
      var banner = GameObject.Find("EventBanner");
      bool bOn = banner != null && banner.activeInHierarchy;
      if (bOn && !BannerSeen && Shot.Add(Round + "banner") && ShotAt < 0) { ShotAt = Time.realtimeSinceStartup + 0.3f; ShotName = "event_banner" + (Round <= 1 ? "" : "_r" + Round) + ".png"; }
      if (bOn != BannerSeen) { BannerSeen = bOn; if (bOn) { BannerAt = Time.realtimeSinceStartup; MusicLogged = false; }
        L("EVENT BANNER " + (bOn ? "shown: '" + string.Join(" / ", banner.GetComponentsInChildren<UnityEngine.UI.Text>().Select(x => x.text)) + "'" : "hidden") + (gm == null ? "" : " event=" + gm.ActiveEvent + " timeRemaining=" + N(gm.TimeRemaining))); }
      if (BannerAt > 0 && !MusicLogged && Time.realtimeSinceStartup >= BannerAt + 1.6f) { MusicLogged = true; L("MUSIC after the event: " + Music()); }
    }
  }
  public static string Arm4(string outDir, string dur) {
    var p = new P4 { Out = outDir, T0 = Time.realtimeSinceStartup, End = float.Parse(dur, System.Globalization.CultureInfo.InvariantCulture) };
    p.L("=== item-4 probe armed (t = real seconds since arm, f = frame, dt = unscaled frame time)");
    UnityEditor.EditorApplication.update += p.Tick; return "armed4"; }
