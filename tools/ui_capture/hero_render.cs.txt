using System.IO;
using System.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.SceneManagement;
using CluckWars.Gameplay;

// Renders one class model, posed by one of its clips, to a transparent PNG. Everything lives in a
// PREVIEW scene (EditorSceneManager.NewPreviewScene) that is closed at the end: nothing is added to,
// or dirties, the open scenes.
public class HeroRender
{
    public static string Render(string cls, string clipName, string tNorm, string yawDeg, string outPath, string sizeStr)
    {
        var reg = AssetDatabase.LoadAssetAtPath<ChickenClassRegistrySO>("Assets/_Game/Data/ChickenClassRegistry.asset");
        if (reg == null) return "ERR no registry";
        var c = (ChickenClass)System.Enum.Parse(typeof(ChickenClass), cls);
        if (!reg.TryGet(c, out var e) || e.ModelPrefab == null) return "ERR no model for " + cls;

        AnimationClip clip = null;
        if (clipName == "Idle") clip = e.Clips.Idle;
        else if (clipName == "Walk") clip = e.Clips.Walk;
        else if (clipName == "Cast") clip = e.Clips.Cast;
        else
        {
            var arch = (CluckWars.Abilities.CastArchetype)System.Enum.Parse(typeof(CluckWars.Abilities.CastArchetype), clipName);
            clip = e.Clips.CastFor(arch);
        }
        if (clip == null) return "ERR no clip " + clipName;

        int size = int.Parse(sizeStr);
        int ss = 2048;                           // supersampled, then auto-cropped + box-filtered to size
        var scene = EditorSceneManager.NewPreviewScene();
        GameObject go = null; var lights = new System.Collections.Generic.List<GameObject>(); Camera cam = null;
        RenderTexture rt = null; Texture2D big = null, small = null;
        try
        {
            go = (GameObject)Object.Instantiate(e.ModelPrefab);
            SceneManager.MoveGameObjectToScene(go, scene);
            go.transform.position = Vector3.zero;
            go.transform.rotation = Quaternion.identity;
            clip.SampleAnimation(go, Mathf.Clamp01(float.Parse(tNorm, System.Globalization.CultureInfo.InvariantCulture)) * clip.length);
            go.transform.rotation = Quaternion.Euler(0f, float.Parse(yawDeg, System.Globalization.CultureInfo.InvariantCulture), 0f);

            // Same tint the match applies (ChickenVisuals.ApplyTint: white -> class tint by strength).
            var tint = Color.Lerp(Color.white, e.TintColor, Mathf.Clamp01(e.TintStrength)); tint.a = 1f;
            var mpb = new MaterialPropertyBlock();
            var rends = go.GetComponentsInChildren<Renderer>();
            foreach (var r in rends)
            {
                r.GetPropertyBlock(mpb);
                mpb.SetColor("_BaseColor", tint); mpb.SetColor("_Color", tint);
                r.SetPropertyBlock(mpb);
                if (r is SkinnedMeshRenderer smr) smr.updateWhenOffscreen = true;
            }

            var b = rends[0].bounds;
            foreach (var r in rends) b.Encapsulate(r.bounds);

            GameObject L(string n, Color col, float intensity, Vector3 euler)
            {
                var lg = new GameObject(n); SceneManager.MoveGameObjectToScene(lg, scene);
                var l = lg.AddComponent<Light>(); l.type = LightType.Directional; l.color = col; l.intensity = intensity;
                l.shadows = LightShadows.None; lg.transform.rotation = Quaternion.Euler(euler); lights.Add(lg); return lg;
            }
            L("Key",  new Color(1.00f, 0.93f, 0.80f), 1.05f, new Vector3(38f, -35f, 0f));
            L("Fill", new Color(0.75f, 0.85f, 1.00f), 0.55f, new Vector3(15f, 140f, 0f));
            L("Rim",  new Color(1.00f, 0.90f, 0.70f), 0.85f, new Vector3(25f, 170f, 0f));
            L("Front",new Color(1.00f, 1.00f, 1.00f), 0.20f, new Vector3(5f, 0f, 0f));

            var cg = new GameObject("HeroCam"); SceneManager.MoveGameObjectToScene(cg, scene);
            cam = cg.AddComponent<Camera>();
            cam.scene = scene;
            cam.clearFlags = CameraClearFlags.SolidColor;
            cam.backgroundColor = new Color(0f, 0f, 0f, 0f);
            cam.allowHDR = false; cam.allowMSAA = false;
            cam.fieldOfView = 22f; cam.nearClipPlane = 0.05f; cam.farClipPlane = 100f;
            var urp = cg.AddComponent<UnityEngine.Rendering.Universal.UniversalAdditionalCameraData>();
            urp.renderPostProcessing = false; urp.antialiasing = UnityEngine.Rendering.Universal.AntialiasingMode.None;
            urp.renderShadows = false;

            // Frame: camera slightly above, looking at the bounds centre from the front; distance so the
            // bounding sphere fills ~86% of the frame height.
            float radius = b.extents.magnitude;
            float dist = radius / Mathf.Sin(Mathf.Deg2Rad * cam.fieldOfView * 0.5f) * 0.93f;
            // Camera on the model's +Z side, 10 degrees above the bounds centre.
            cam.transform.position = b.center + (Quaternion.Euler(-10f, 0f, 0f) * Vector3.forward) * dist;
            cam.transform.LookAt(b.center);

            rt = new RenderTexture(ss, ss, 24, RenderTextureFormat.ARGB32) { antiAliasing = 1 };
            rt.Create();
            cam.targetTexture = rt;
            cam.Render();

            RenderTexture.active = rt;
            big = new Texture2D(ss, ss, TextureFormat.RGBA32, false, false);
            big.ReadPixels(new Rect(0, 0, ss, ss), 0, 0); big.Apply();
            RenderTexture.active = null;

            // Auto-crop on the alpha bounding box (renderer bounds of a skinned mesh are loose), then
            // box-filter down to size with premultiplied colour so edges don't fringe. Square crop,
            // centred horizontally, feet on a common baseline 5% above the bottom edge.
            var src = big.GetPixels32();
            int minX = ss, minY = ss, maxX = -1, maxY = -1;
            for (int y = 0; y < ss; y++)
            for (int x = 0; x < ss; x++)
                if (src[y * ss + x].a > 8) { if (x < minX) minX = x; if (x > maxX) maxX = x; if (y < minY) minY = y; if (y > maxY) maxY = y; }
            if (maxX < 0) return "ERR empty render";
            float side = Mathf.Max(maxX - minX, maxY - minY) * 1.10f;
            float cx = (minX + maxX) * 0.5f;
            float x0 = cx - side * 0.5f, y0 = minY - side * 0.05f;   // texture y is bottom-up
            small = new Texture2D(size, size, TextureFormat.RGBA32, false, false);
            var dst = new Color32[size * size];
            float step = side / size;
            for (int y = 0; y < size; y++)
            for (int x = 0; x < size; x++)
            {
                float r = 0, g = 0, bl = 0, a = 0; int n = 0;
                int sx0 = Mathf.FloorToInt(x0 + x * step), sx1 = Mathf.Max(sx0 + 1, Mathf.FloorToInt(x0 + (x + 1) * step));
                int sy0 = Mathf.FloorToInt(y0 + y * step), sy1 = Mathf.Max(sy0 + 1, Mathf.FloorToInt(y0 + (y + 1) * step));
                for (int sy = sy0; sy < sy1; sy++)
                for (int sx = sx0; sx < sx1; sx++)
                {
                    n++;
                    if (sx < 0 || sy < 0 || sx >= ss || sy >= ss) continue;
                    var p = src[sy * ss + sx];
                    float pa = p.a / 255f; r += p.r * pa; g += p.g * pa; bl += p.b * pa; a += pa;
                }
                dst[y * size + x] = a > 0.0001f
                    ? new Color32((byte)Mathf.Clamp(r / a, 0, 255), (byte)Mathf.Clamp(g / a, 0, 255), (byte)Mathf.Clamp(bl / a, 0, 255), (byte)Mathf.Clamp(a / n * 255f, 0, 255))
                    : new Color32(0, 0, 0, 0);
            }
            small.SetPixels32(dst); small.Apply();
            Directory.CreateDirectory(Path.GetDirectoryName(outPath));
            File.WriteAllBytes(outPath, small.EncodeToPNG());
            int opaque = dst.Count(p => p.a > 200);
            return $"ok {outPath} clip={clip.name} len={clip.length:0.00} opaque={opaque} bounds={b.size}";
        }
        finally
        {
            if (cam != null) cam.targetTexture = null;
            if (rt != null) { rt.Release(); Object.DestroyImmediate(rt); }
            if (big != null) Object.DestroyImmediate(big);
            if (small != null) Object.DestroyImmediate(small);
            EditorSceneManager.ClosePreviewScene(scene);   // destroys go, lights and camera with it
        }
    }
}
